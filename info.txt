
---Roles -----

CREATE TABLE tr_roles ( id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(100) UNIQUE NOT NULL, description VARCHAR(255), status TINYINT DEFAULT 1 COMMENT '1 = Active, 0 = Inactive', created_by INT NULL, updated_by INT NULL, created_at DATETIME DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, FOREIGN KEY (created_by) REFERENCES user_details(user_id) ON DELETE SET NULL, FOREIGN KEY (updated_by) REFERENCES user_details(user_id) ON DELETE SET NULL )



----Permissions------

CREATE TABLE tr_permissions ( id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(100) UNIQUE NOT NULL, description VARCHAR(255), status TINYINT DEFAULT 1 COMMENT '1 = Active, 0 = Inactive', created_by INT NULL, updated_by INT NULL, created_at DATETIME DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, CONSTRAINT fk_permissions_created_by FOREIGN KEY (created_by) REFERENCES user_details(user_id) ON DELETE SET NULL, CONSTRAINT fk_permissions_updated_by FOREIGN KEY (updated_by) REFERENCES user_details(user_id) ON DELETE SET NULL ) ENGINE=InnoDB


----User Roles ------

CREATE TABLE tr_user_roles ( id INT AUTO_INCREMENT PRIMARY KEY, user_id INT NOT NULL, role_id INT NOT NULL, status TINYINT DEFAULT 1 COMMENT '1 = Active, 0 = Inactive', assigned_by INT NULL, assigned_at DATETIME DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, UNIQUE KEY unique_user_role (user_id, role_id), FOREIGN KEY (user_id) REFERENCES user_details(user_id) ON DELETE CASCADE, FOREIGN KEY (role_id) REFERENCES tr_roles(id) ON DELETE CASCADE, FOREIGN KEY (assigned_by) REFERENCES user_details(user_id) ON DELETE SET NULL ) ENGINE=InnoDB



----Role Permissions -----

CREATE TABLE tr_role_permissions ( id INT AUTO_INCREMENT PRIMARY KEY, role_id INT NOT NULL, permission_id INT NOT NULL, status TINYINT DEFAULT 1 COMMENT '1 = Active, 0 = Inactive', assigned_by INT NULL, assigned_at DATETIME DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, UNIQUE KEY unique_role_permission (role_id, permission_id), FOREIGN KEY (role_id) REFERENCES tr_roles(id) ON DELETE CASCADE, FOREIGN KEY (permission_id) REFERENCES tr_permissions(id) ON DELETE CASCADE, FOREIGN KEY (assigned_by) REFERENCES user_details(user_id) ON DELETE SET NULL ) ENGINE=InnoDB
